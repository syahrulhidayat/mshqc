setopt interactivecomments

# 1. Tutup Firefox dan hapus file pengunci profil (.parentlock) akibat pkill -9
pkill -f firefox 2>/dev/null
sleep 2

for dir in ~/.mozilla/firefox/*.default*(N) ~/.var/app/org.mozilla.firefox/.mozilla/firefox/*.default*(N); do
    rm -f "$dir/lock" "$dir/.parentlock"
    
    # 2. Kembalikan jaringan Firefox ke Default Sistem (agar langsung memakai koneksi WARP sistem)
    cat << 'PREFS' > "$dir/user.js"
user_pref("network.proxy.type", 0);
user_pref("network.trr.mode", 0);
user_pref("network.trr.uri", "");
user_pref("network.trr.bootstrapAddr", "");
user_pref("network.dns.disableIPv6", false);
user_pref("network.http.altsvc.enabled", true);
user_pref("network.http.http3.enabled", true);

// Pertahankan pengaturan Region & GPS Amerika Serikat
user_pref("geo.enabled", true);
user_pref("geo.provider.use_geoclue", false);
user_pref("geo.provider.testing", true);
user_pref("geo.provider.network.url", "data:application/json,{\"location\": {\"lat\": 40.7590, \"lng\": -73.9845}, \"accuracy\": 20.0}");
user_pref("browser.search.region", "US");
user_pref("intl.accept_languages", "en-US, en");
PREFS
done

# 3. Kembalikan ke direktori home dan jalankan Firefox secara normal
cd ~
TZ=America/New_York firefox "https://www.google.com/safesearch?hl=en&gl=us" &
